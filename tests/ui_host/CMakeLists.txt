cmake_minimum_required(VERSION 3.16)
project(wordcards_ui_host C)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/managed_components/lvgl__lvgl" lvgl)
file(GLOB FONTS "${ROOT}/assets/fonts/wc_font_*.c")
add_executable(ui_host render.c "${ROOT}/main/wc_ui.c" "${ROOT}/main/wc_model.c" "${ROOT}/main/wc_data.c" "${ROOT}/assets/images/wc_portraits.c" "${ROOT}/components/bsp/src/bsp_display_rounding.c" ${FONTS})
target_include_directories(ui_host PRIVATE . "${ROOT}/main" "${ROOT}/components/bsp/src")
target_link_libraries(ui_host lvgl m)
